def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def load(path):
    return json.loads(Path(path).read_text(encoding="utf-8"))


def exact(left, right):
    return json.dumps(left, sort_keys=True, separators=(",", ":"), default=sorted) == json.dumps(
        right, sort_keys=True, separators=(",", ":"), default=sorted
    )


def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":")).encode("ascii")).hexdigest()


def issha(value):
    return isinstance(value, str) and len(value) == 64 and all(char in "0123456789abcdef" for char in value)


def finite(value, positive=False):
    return type(value) in (int, float) and math.isfinite(value) and (value > 0 if positive else value >= 0)


class Audit:
    def __init__(self):
        self.checks = 0

    def require(self, condition, label):
        self.checks += 1
        if not condition:
            raise ValueError(label)

    def same(self, left, right, label):
        self.require(exact(left, right), label)


def owned(root, relative):
    if (
        not isinstance(relative, str)
        or not relative
        or "\\" in relative
        or PurePosixPath(relative).is_absolute()
        or PureWindowsPath(relative).drive
        or ".." in PurePosixPath(relative).parts
    ):
        raise ValueError("invalid relative evidence path")
    path = (root / relative).resolve(strict=True)
    if not path.is_relative_to(root) or not path.is_file():
        raise ValueError("evidence path escapes its root")
    return path


def expected_records(repo, deployed, audit):
    results = repo / "benchmarks/results/multi_graph_structural"
    trace_path = results / "structural-dynamic-trace-v1.json"
    selection_path = results / "q4-tile-full-replay-v8-queue0.rank0.json"
    audit.same(sha(trace_path), deployed[trace_path.name], "deployed canonical trace bytes")
    audit.same(sha(selection_path), deployed[selection_path.name], "deployed default selection bytes")
    document, selected = load(trace_path), load(selection_path)
    trace = document["trace"]
    normalized = dict(
        trace, initial_prefix_by_request={int(key): value for key, value in trace["initial_prefix_by_request"].items()}
    )
    audit.same(digest(normalized), CANONICAL, "canonical independent digest")
    audit.same(document["trace_sha256"], CANONICAL, "canonical declared digest")
    audit.same(selected["trace_sha256"], CANONICAL, "selection digest")
    selection = selected["selection"]
    sizes, dispatch = selection["capture_sizes"], selection["dispatched_tokens"]
    audit.same(sizes, sorted(set(sizes)), "sorted original default catalog")
    audit.same(
        selection["config_source"], "installed EngineArgs defaults; no capture sizes supplied", "default catalog origin"
    )
    audit.same([row["step"] for row in trace["steps"]], list(range(128)), "canonical chronological steps")
    maximum = [trace["initial_prefix_by_request"][str(owner)] for owner in range(128)]
    for step in trace["steps"]:
        for owner, length in zip(step["request_ids"], step["seq_lengths"], strict=True):
            maximum[owner] = max(maximum[owner], length)
    pages, cursor = [], 1
    for length in maximum:
        count = (length + 127) // 128
        pages.append(list(range(cursor, cursor + count)))
        cursor += count
    width = max(map(len, pages))
    audit.same([cursor, width], [1293, 15], "owner reservation geometry")
    prefix = [trace["initial_prefix_by_request"][str(owner)] for owner in range(128)]
    pending, previous, seen, records = [1] * 128, set(), set(), []
    for index, step in enumerate(trace["steps"]):
        owners, lengths, batch = step["request_ids"], step["seq_lengths"], step["batch"]
        audit.require(
            batch == len(owners) == len(lengths) and len(set(owners)) == batch, f"step {index}: unique live owners"
        )
        audit.same(step["first_use"], [owner not in seen for owner in owners], f"step {index}: lifecycle first use")
        audit.same(
            step["accepted_from_previous"], [pending[owner] for owner in owners], f"step {index}: accepted lengths"
        )
        for owner in owners:
            if owner in seen:
                prefix[owner] += pending[owner]
        audit.same(lengths, [prefix[owner] + 4 for owner in owners], f"step {index}: exact lengths")
        capacity = sizes[bisect.bisect_left(sizes, batch * 4)]
        audit.same(capacity, dispatch[index], f"step {index}: actual smallest default bucket")
        physical, padding, positions, slots = capacity // 4, capacity // 4 - batch, [], []
        for owner, length in zip(owners, lengths, strict=True):
            positions.extend(range(length - 4, length))
            slots.extend(pages[owner][position // 128] * 128 + position % 128 for position in range(length - 4, length))
        current = set(owners)
        records.append(
            dict(
                step=index,
                batch=batch,
                live_tokens=batch * 4,
                physical_tokens=capacity,
                default_capacity=capacity,
                physical_batch=physical,
                padding_requests=padding,
                request_ids=owners,
                owner_keys=[[owner, 0] for owner in owners] + [None] * padding,
                owner_active=[owner in current for owner in range(128)],
                q_lengths=[4] * physical,
                actual_seq_qlen=list(range(4, capacity + 1, 4)),
                kv_lengths=lengths + [0] * padding,
                block_table=[pages[owner] + [0] * (width - len(pages[owner])) for owner in owners]
                + [[0] * width for _ in range(padding)],
                positions=positions + [0] * (padding * 4),
                slot_mapping=slots + [-1] * (padding * 4),
                active_tokens=[True] * (batch * 4) + [False] * (padding * 4),
                first_use=step["first_use"],
                accepted_from_previous=step["accepted_from_previous"],
                accepted_after_step=step["accepted_after_step"],
                admitted=[owner for owner in owners if owner not in seen],
                resumed=[owner for owner in owners if owner in seen and owner not in previous],
                departed=sorted(previous - current),
                retired=[],
                input_seed=step["input_seed"],
                page_count=cursor,
            )
        )
        for owner, accepted in zip(owners, step["accepted_after_step"], strict=True):
            audit.require(type(accepted) is int and 1 <= accepted <= 4, "accepted query prefix bounds")
            pending[owner] = accepted
        previous = current
        seen.update(current)
    first = {}
    for record in records:
        first.setdefault(record["physical_tokens"], record)
    audit.same(list(first), ORDER, "independent first-use catalog order")
    audit.same([row["step"] for row in first.values()], STEPS, "independent first-use steps")
    return records, first, selection, dict(pages=pages, maximum_lengths=maximum, page_count=cursor, width=width)


def plan_check(plan, record, mode, audit):
    words = plan["words"]
    audit.require(
        isinstance(words, list)
        and len(words) == 1024
        and all(type(value) is int and 0 <= value < 2**31 for value in words),
        "complete native plan words",
    )
    audit.same(hashlib.sha256(struct.pack("<1024i", *words)).hexdigest(), plan["sha256"], "native plan raw byte hash")
    base = words[576:586]
    maximum, batch = max(record["kv_lengths"]) or 1024, record["physical_batch"]
    audit.same(
        base[:3] + base[6:], [512, min(512, maximum), 8, 4, maximum, batch, batch], "native exact current geometry"
    )
    audit.require(type(base[4]) is int and 1 <= base[4] <= 20, "native active core count")
    audit.require(plan["current_exact_base_verified"] is True, "runtime current-base validation")
    if mode == "whole":
        audit.same([base[3], base[5]], [0, 0], "whole plan excludes reduction groups and cores")
        expected, begin, loads = [0] * 1024, 0, []
        expected[576:586] = base
        for core in range(base[4]):
            slot = words[core * 10 : (core + 1) * 10]
            end = slot[1]
            audit.require(
                slot[0] == 1 and slot[2:] == [0] * 8 and begin < end <= batch * 2, "whole advancing non-FD core range"
            )
            load = 0
            for head in range(begin, end):
                length = record["kv_lengths"][head // 2]
                full, tail = divmod(length, base[1])
                load += full * (12 + 10 * ((base[1] + 63) // 64)) + (12 + 10 * ((tail + 63) // 64) if tail else 0)
            audit.require(load > 0 or not any(record["kv_lengths"]), "whole active core contains real work")
            loads.append(load)
            expected[core * 10 : (core + 1) * 10] = slot
            begin = end
        audit.same(begin, batch * 2, "whole complete physical head coverage")
        for vector in range(2 * base[4]):
            expected[360 + 3 * vector] = 1
        audit.same(words, expected, "whole unused and FD words are all zero")
        audit.same(
            plan["whole_coverage"],
            dict(
                used_cores=base[4],
                physical_heads=batch * 2,
                attention_tiles=sum((length + base[1] - 1) // base[1] * 2 for length in record["kv_lengths"]),
                core_costs=loads,
                cost_model="abstract_6_10_upper_bound",
            ),
            "independent whole coverage result",
        )
    return words


def tensor_identity(item, shape, audit, dtype="torch.bfloat16"):
    audit.same(item["shape"], shape, "tensor exact shape")
    audit.same(item["dtype"], dtype, "tensor exact dtype")
    audit.require(issha(item["sha256"]), "tensor full byte SHA256")


def input_check(report, audit):
    value = report["input_identity"]
    audit.same(value, report["post_input_identity"], "resident complete QKV unchanged")
    audit.same(report["mask_identity"], report["post_mask_identity"], "resident complete mask unchanged")
    audit.same(value["sha256"], digest(value["occurrences"]), "resident aggregate hash")
    audit.same(len(value["occurrences"]), 16, "sixteen private QKV occurrences")
    shapes = dict(query=[512, 12, 256], key=[1293, 128, 2, 256], value=[1293, 128, 2, 256])
    for occurrence in value["occurrences"]:
        audit.same(set(occurrence), set(shapes), "QKV tensor identity keys")
        for key, shape in shapes.items():
            tensor_identity(occurrence[key], shape, audit)
    tensor_identity(report["mask_identity"], [2048, 2048], audit, "torch.bool")
    ranges = report["private_storage_ranges"]
    audit.same(len(ranges), 48, "all private QKV storage ranges")
    audit.same(
        {(row["occurrence"], row["name"]) for row in ranges},
        {(occurrence, name) for occurrence in range(16) for name in shapes},
        "private storage identities",
    )
    for row in ranges:
        audit.require(type(row["start"]) is int and row["start"] > 0, "private positive base address")
        audit.same(row["end"] - row["start"], math.prod(shapes[row["name"]]) * 2, "private storage exact bytes")
    ordered = sorted((row["start"], row["end"]) for row in ranges)
    audit.require(all(left[1] <= right[0] for left, right in zip(ordered, ordered[1:])), "private QKV no overlap")


def catalog_storage_check(report, captures, audit):
    keys = {str(row["physical_tokens"]) for row in captures}
    for name in ("frame_query_identity", "post_frame_query_identity", "captured_addresses", "prepared_descriptors"):
        audit.same(set(report[name]), keys, f"complete catalog {name} coverage")
    audit.same(
        report["frame_query_identity"], report["post_frame_query_identity"], "all retained query frames unchanged"
    )
    ranges = []
    for record in captures:
        capacity, batch = record["physical_tokens"], record["physical_batch"]
        for frame in report["frame_query_identity"][str(capacity)]:
            tensor_identity(frame, [capacity, 12, 256], audit)
        audit.same(len(report["frame_query_identity"][str(capacity)]), 16, "sixteen per-bucket query frames")
        shapes = dict(
            q_lengths=[batch], kv_lengths=[batch], block_table=[batch, 15], atten_mask=[2048, 2048], metadata=[1024]
        )
        shapes.update({f"{name}{index}": [capacity, 12, 256] for name in ("query", "output") for index in range(16)})
        saved = report["captured_addresses"][str(capacity)]
        audit.same(set(saved), set(shapes), "exact retained descriptor set")
        for name, shape in shapes.items():
            descriptor = saved[name]
            audit.same(descriptor["shape"], shape, "captured descriptor shape")
            audit.same(
                descriptor["stride"],
                [math.prod(shape[i + 1 :]) for i in range(len(shape))],
                "captured contiguous stride",
            )
            audit.require(type(descriptor["pointer"]) is int and descriptor["pointer"] > 0, "captured positive pointer")
            size = 8 if name in ("q_lengths", "kv_lengths") else 4 if name in ("block_table", "metadata") else 2
            if name != "atten_mask":
                ranges.append((descriptor["pointer"], descriptor["pointer"] + math.prod(shape) * size))
        count = 17 if capacity in CAPS else 0
        prep = report["prepared_descriptors"][str(capacity)]
        wanted = dict(
            mode="sink",
            physical_tokens=capacity,
            physical_requests=batch,
            host_pinned=dict.fromkeys(("q_lengths", "kv_lengths", "block_table"), True),
            stable_host_allocations=True,
            stable_device_allocations=True,
            host_allocation_count=3,
            device_allocation_count=0,
            setup_copies=dict(q_lengths=1, kv_lengths=0, block_table=0),
            step_copies=dict(q_lengths=0, kv_lengths=count, block_table=count),
            prepared_steps=count,
            completed_steps=count,
            inflight=False,
            q4_invariant=dict(query_length=4, cumulative_q=list(range(4, capacity + 1, 4))),
        )
        for key, value in wanted.items():
            audit.same(prep[key], value, f"capacity {capacity}: persistent descriptor {key}")
        for key in ("q_lengths", "kv_lengths", "block_table"):
            for field in ("pointer", "shape", "stride"):
                audit.same(
                    prep["device"][key][field], saved[key][field], "prepared source targets exact captured descriptor"
                )
    ordered = sorted(ranges)
    audit.require(
        all(left[1] <= right[0] for left, right in zip(ordered, ordered[1:])),
        "independent bucket storage does not overlap",
    )

def stats(values):
    ordered = sorted(values)

    def percentile(fraction):
        position = (len(ordered) - 1) * fraction
        lower = int(position)
        upper = min(lower + 1, len(ordered) - 1)
        return ordered[lower] + (ordered[upper] - ordered[lower]) * (position - lower)

    return dict(
        count=len(values),
        mean_us=statistics.mean(values),
        minimum_us=min(values),
        maximum_us=max(values),
        p50_us=percentile(0.5),
        p95_us=percentile(0.95),
        p99_us=percentile(0.99),
    )


def summaries(indexed):
    cells, contexts, policies, interactions = [], [], [], []
    for mode, count in CELLS:
        for capacity in CAPS:
            selected = [indexed[mode, count, capacity, repeat] for repeat in range(10)]
            cells.append(
                dict(
                    mode=mode,
                    catalog_count=count,
                    capacity=capacity,
                    metrics={metric: stats([row[metric] for row in selected]) for metric in METRICS},
                )
            )

    def comparison(left_mode, left_count, right_mode, right_count, capacity):
        result = {}
        for metric in METRICS:
            left = [indexed[left_mode, left_count, capacity, repeat][metric] for repeat in range(10)]
            right = [indexed[right_mode, right_count, capacity, repeat][metric] for repeat in range(10)]
            deltas = [a - b for a, b in zip(left, right, strict=True)]
            result[metric] = dict(
                left=stats(left),
                right=stats(right),
                left_minus_right=stats(deltas),
                left_reduction_percent=100 * sum(deltas) / sum(left),
                right_faster_count=sum(value > 0 for value in deltas),
            )
        return result

    for capacity in CAPS:
        for mode in ("upstream", "whole"):
            contexts.append(
                dict(
                    mode=mode,
                    capacity=capacity,
                    left_catalog=2,
                    right_catalog=17,
                    metrics=comparison(mode, 2, mode, 17, capacity),
                )
            )
        for count in (2, 17):
            policies.append(
                dict(
                    capacity=capacity,
                    catalog_count=count,
                    left_mode="upstream",
                    right_mode="whole",
                    metrics=comparison("upstream", count, "whole", count, capacity),
                )
            )
        metrics = {}
        for metric in METRICS:
            values = [
                (indexed["upstream", 17, capacity, repeat][metric] - indexed["whole", 17, capacity, repeat][metric])
                - (indexed["upstream", 2, capacity, repeat][metric] - indexed["whole", 2, capacity, repeat][metric])
                for repeat in range(10)
            ]
            metrics[metric] = stats(values)
        interactions.append(
            dict(capacity=capacity, definition="(upstream17 - whole17) - (upstream2 - whole2)", metrics=metrics)
        )
    return dict(per_cell_capacity=cells, catalog_context=contexts, native_policy=policies, interaction=interactions)

        suite["context_screen"] is True and suite["performance_measured"] is True, "bounded context performance screen"

        "stable_performance_gain_established",

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("bundle", type=Path, help="catalog-bridge-suite.json, or its collected evidence directory")
    parser.add_argument("--repo", type=Path, default=Path(__file__).resolve().parents[2] / "work/vllm-ascend")
    parser.add_argument("--output", type=Path)
    args = parser.parse_args()
    path = args.bundle.resolve(strict=True)
    if path.is_dir():
        path = path / "catalog-bridge-suite.json"
    output = args.output.resolve() if args.output else path.parent / "catalog-bridge-independent-audit.json"
    protected = {owned(path.parent, item["path"]) for item in load(path.parent / "manifest.json")}
    if output in protected or output == path.parent / "manifest.json":
        parser.error("audit output cannot overwrite raw manifested evidence")
    audit = Audit()
    try:
        result = audit_bundle(path, args.repo.resolve(strict=True), audit)
    except (ValueError, KeyError, TypeError, OSError, IndexError, struct.error) as error:
        result = dict(
            qualified=False,
            independent_checks=audit.checks,
            error=str(error),
            source=dict(bundle=str(path), auditor_sha256=sha(__file__)),
        )
    output.write_text(json.dumps(result, indent=2, allow_nan=False) + "\n", encoding="ascii")
    print(
        json.dumps(
            dict(output=str(output), qualified=result["qualified"], checks=audit.checks, error=result.get("error"))
        )
    )
    return 0 if result["qualified"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
