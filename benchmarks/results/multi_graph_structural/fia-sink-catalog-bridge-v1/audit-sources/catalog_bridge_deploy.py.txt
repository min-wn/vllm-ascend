"""Deploy a fixed, isolated four-cell screen after local validation."""

import base64
import argparse
import shlex
import subprocess
from pathlib import Path

repo = Path(__file__).resolve().parents[2] / "work/vllm-ascend"
parser = argparse.ArgumentParser()
parser.add_argument("--version", type=int, choices=(1, 2), required=True)
parser.add_argument("--probe-sha256", required=True)
args = parser.parse_args()
assert len(args.probe_sha256) == 64 and all(value in "0123456789abcdef" for value in args.probe_sha256)
paths = list((repo / "benchmarks").glob("aclgraph_fia_sink_stage*.py"))
paths += list((repo / "benchmarks").glob("aclgraph_fia_sink_consumer*.py"))
paths += list((repo / "benchmarks").glob("aclgraph_fia_sink_catalog_bridge*.py"))
paths += [repo / "benchmarks" / name for name in (
    "multi_graph_gdn_q4_full_metadata.py", "multi_graph_gdn_q4_full_loader.py", "multi_graph_dynamic_trace.py",
)]
paths += [repo / "benchmarks/results/multi_graph_structural" / name for name in (
    "structural-dynamic-trace-v1.json", "q4-tile-full-replay-v8-queue0.rank0.json",
)]
files = {path.name: base64.b64encode(path.read_bytes()).decode() for path in paths}
script = "FILES=" + repr(files) + "\nVERSION=" + repr(args.version) + "\nPROBE_SHA=" + repr(args.probe_sha256) + "\n" + '''import base64,hashlib,json,subprocess,sys
from pathlib import Path
root=Path('/workspace/aclgraph-sink-stage2-d256-bec19b8dea1e')
assert json.loads((root/'stage5-native-gate-v3/stage5-native-gate.json').read_text())['safety_gate_passed'] is True
assert json.loads((root/'stage5-regression-v3/status.json').read_text())['state']=='completed'
info=subprocess.run(['npu-smi','info'],text=True,stdout=subprocess.PIPE,check=True).stdout
assert 'No running processes found in NPU 0' in info,info
tools=root/f'catalog-bridge-tools-v{VERSION}'
tools.mkdir(exist_ok=False)
(tools/'device-before.txt').write_text(info)
for name,data in FILES.items():
    assert Path(name).name==name
    (tools/name).write_bytes(base64.b64decode(data))
sha={name:hashlib.sha256((tools/name).read_bytes()).hexdigest() for name in FILES}
assert sha['aclgraph_fia_sink_consumer_probe.py']=='b7472c5ccbdb37d4f5ea1693f2abc5145a71abb773d1e09ea788bf5daca964c9'
assert sha['aclgraph_fia_sink_catalog_bridge_probe.py']==PROBE_SHA
assert sha['aclgraph_fia_sink_catalog_bridge_suite.py']=='d63a89946c8fc4f7f0d8eeaaa05df991f805f4a1e4356ec71a5d03a24d2b865a'
(tools/'deployed-tools.json').write_text(json.dumps(sha,indent=2)+'\\n')
command=[sys.executable,str(tools/'aclgraph_fia_sink_catalog_bridge_suite.py'),'--root',str(root),'--upstream-vendor',str(root/'opp/packages/vendors/thesis_fia_stage2_transformer'),'--whole-vendor',str(root/'stage5-no-fd/relinked-v3/opp/packages/vendors/materialized-baseline-v3'),'--output-dir',str(root/f'catalog-bridge-v{VERSION}'),'--trace',str(tools/'structural-dynamic-trace-v1.json'),'--selection',str(tools/'q4-tile-full-replay-v8-queue0.rank0.json'),'--timeout','1800']
with (tools/'launch.log').open('xb') as log:
    child=subprocess.Popen(command,stdout=log,stderr=subprocess.STDOUT,stdin=subprocess.DEVNULL,start_new_session=True)
print(json.dumps(dict(pid=child.pid,command=command,status=str(root/f'catalog-bridge-v{VERSION}/catalog-bridge-suite.json'))))
'''
result = subprocess.run(
    ["ssh", "-o", "BatchMode=yes", "-o", "ConnectTimeout=20", "vllm-ascend-030", "bash -lc " + shlex.quote("source /etc/profile.d/ascend-multigraph.sh; python -")],
    input=script.encode(), timeout=45,
)
raise SystemExit(result.returncode)
